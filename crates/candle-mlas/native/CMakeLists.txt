cmake_minimum_required(VERSION 3.25)
project(candle_mlas LANGUAGES C CXX ASM)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "This standalone recipe is qualified only for Linux x86_64; port/requalify other platforms explicitly")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(onnxruntime_ORT_MINIMAL_BUILD ON)
set(onnxruntime_BUILD_SHARED_LIB ON)
set(ONNXRUNTIME_ROOT "${ORT_SOURCE}/onnxruntime")
if(NOT EXISTS "${ORT_SOURCE}/cmake/onnxruntime_mlas.cmake")
  message(FATAL_ERROR "ORT_SOURCE must name the verified v1.30.0 source tree")
endif()
function(onnxruntime_add_static_library name)
  add_library(${name} STATIC ${ARGN})
  target_compile_definitions(${name} PRIVATE BUILD_MLAS_NO_ONNXRUNTIME)
  target_include_directories(${name} PRIVATE "${ONNXRUNTIME_ROOT}" "${ORT_SOURCE}/include/onnxruntime" "${SAFEINT_SOURCE}" "${GSL_SOURCE}/include")
  target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-include;${CMAKE_CURRENT_SOURCE_DIR}/standalone_prefix.h>")
endfunction()
function(onnxruntime_add_include_to_target name)
  foreach(dependency IN LISTS ARGN)
    if(TARGET ${dependency})
      target_link_libraries(${name} PRIVATE ${dependency})
    endif()
  endforeach()
endfunction()
include("${ORT_SOURCE}/cmake/onnxruntime_mlas.cmake")
# The upstream platform table still references these kernels in a minimal build.
# Include the real definitions so the standalone shared library has no unresolved symbols.
target_sources(onnxruntime_mlas PRIVATE
  "${ONNXRUNTIME_ROOT}/core/mlas/lib/q4_dq.cpp"
  "${ONNXRUNTIME_ROOT}/core/mlas/lib/q4gemm.cpp"
  "${ONNXRUNTIME_ROOT}/core/mlas/lib/q4gemm_avx512.cpp")
set_source_files_properties("${ONNXRUNTIME_ROOT}/core/mlas/lib/q4gemm_avx512.cpp" PROPERTIES
  COMPILE_FLAGS "-mfma -mavx512vnni -mavx512bw -mavx512dq -mavx512vl -mavx512f")
find_package(OpenMP REQUIRED)
add_library(candle_mlas SHARED bridge.cpp stacktrace.cpp)
target_include_directories(candle_mlas PRIVATE "${ONNXRUNTIME_ROOT}/core/mlas/inc")
target_link_libraries(candle_mlas PRIVATE ${ONNXRUNTIME_MLAS_LIBS} OpenMP::OpenMP_CXX)
option(MLAS_TARGET_NATIVE "Compile bridge for current host; rebuild for other CPUs" OFF)
set(BRIDGE_OPTIONS -O3 -ffp-contract=off)
if(MLAS_TARGET_NATIVE)
  list(APPEND BRIDGE_OPTIONS -march=native)
endif()
target_compile_options(candle_mlas PRIVATE ${BRIDGE_OPTIONS})
# The qualified FP32 MLAS paths execute static text, not code on the stack.
# Older linkers otherwise inherit RWE from upstream assembly lacking GNU-stack notes.
target_link_options(candle_mlas PRIVATE -Wl,--no-undefined -Wl,-z,noexecstack)

if(VELA_KERNEL_SOURCE)
  add_library(vela_mlas SHARED "${VELA_KERNEL_SOURCE}")
  target_include_directories(vela_mlas PRIVATE "${ONNXRUNTIME_ROOT}/core/mlas/inc")
  target_link_libraries(vela_mlas PRIVATE candle_mlas OpenMP::OpenMP_CXX)
  target_compile_options(vela_mlas PRIVATE ${BRIDGE_OPTIONS})
  target_link_options(vela_mlas PRIVATE -Wl,--no-undefined -Wl,-z,noexecstack)
  set_target_properties(vela_mlas PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
